cmake_minimum_required(VERSION 3.16)
project(DarwinEES VERSION 1.0.0 LANGUAGES CXX)
option(DARWIN_EES_BUILD_TESTS "Build tests" ON)
add_library(darwin_ees_core src/Geometry.cpp src/Entity.cpp src/Herbivore.cpp src/Predator.cpp src/Resources.cpp src/Ecosystem.cpp src/CsvExporter.cpp)
target_include_directories(darwin_ees_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(darwin_ees_core PUBLIC cxx_std_17)
if(MSVC)
 target_compile_options(darwin_ees_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(darwin_ees_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(darwin_ees src/main.cpp)
target_link_libraries(darwin_ees PRIVATE darwin_ees_core)
if(DARWIN_EES_BUILD_TESTS)
 enable_testing()
 add_executable(darwin_ees_tests tests/simulation_tests.cpp)
 target_link_libraries(darwin_ees_tests PRIVATE darwin_ees_core)
 add_test(NAME darwin_ees_tests COMMAND darwin_ees_tests)
endif()
